local function killSubmitDebounce(fn)
    if not (fn and setupv and getupvalues) then return end
    local ok, ups = pcall(getupvalues, fn)
    if ok and type(ups) == "table" then
        for i, v in pairs(ups) do
            if type(v) == "boolean" then
                pcall(setupv, fn, i, false)
            end
        end
    end
end
local function tryFireConnections(signal, ...)
    if not (getconns and signal) then return false end
    local ok, conns = pcall(getconns, signal)
    if not ok or type(conns) ~= "table" or #conns == 0 then return false end
    local fired = false
    local args = table.pack(...)
    for _, c in ipairs(conns) do
        local fn = nil
        pcall(function() fn = c.Function end)
        killSubmitDebounce(fn)
        local thisOk = pcall(function()
            if c.Enabled ~= false then
                c:Fire(table.unpack(args, 1, args.n))
            end
        end)
        if not thisOk then
            thisOk = pcall(function()
                if c.Enabled ~= false then
                    c:Fire()
                end
            end)
        end
        fired = fired or thisOk
    end
    return fired
end
local function resolveAceRedeemRemote()
    if RedeemRemote and RedeemRemote.Parent then return RedeemRemote end
    if not Net then return nil end
    local ok, api = pcall(require, Net)
    if ok and type(api) == "table" then
        local rok, rf = pcall(function()
            return api:RemoteFunction(REDEEM_GUID)
        end)
        if rok and typeof(rf) == "Instance" then
            RedeemRemote = rf
        end
    end
    return RedeemRemote
end
local function looksLikeRedeemButton(obj)
    if not obj then return false end
    if not (obj:IsA("TextButton") or obj:IsA("ImageButton")) then return false end
    local name = tostring(obj.Name or ""):lower()
    local text = ""
    pcall(function()
        text = tostring(obj.Text or ""):lower()
    end)
    return name:find("redeem", 1, true)
        or name:find("submit", 1, true)
        or name:find("claim", 1, true)
        or text:find("redeem", 1, true)
        or text:find("submit", 1, true)
        or text:find("claim", 1, true)
end
local function findRedeemButton(box)
    local roots = {}
    local parent = box and box.Parent
    if parent then roots[#roots + 1] = parent end
    if parent and parent.Parent then roots[#roots + 1] = parent.Parent end
    local codesGui = playerGui:FindFirstChild("Codes")
    if codesGui then roots[#roots + 1] = codesGui end
    local seen = {}
    for _, root in ipairs(roots) do
        if root and not seen[root] then
            seen[root] = true
            for _, obj in ipairs(root:GetDescendants()) do
                if obj ~= box and looksLikeRedeemButton(obj) then
                    return obj
                end
            end
        end
    end
    return nil
end
local function submitByRedeemButton(box)
    local btn = findRedeemButton(box)
    if not btn then return false end
    local ok = pcall(function()
        btn:Activate()
    end)
    if ok then return true end
    if tryFireConnections(btn.MouseButton1Click) then return true end
    local activated = btn:IsA("GuiButton") and btn.Activated or nil
    if activated and tryFireConnections(activated) then return true end
    return false
end
local function submitByRedeemRemote(box)
    local rf = resolveAceRedeemRemote()
    if not rf then return false end
    local submitText = (box and box.Text) or _lastNonBlankBoxText or ""
    if submitText == "" then return false end
    return pcall(function()
        rf:InvokeServer(submitText)
    end)
end

local function loadConfig()
    if not isfile then return end
    if not isfile(CONFIG_FILE) then return end
    local raw = readfile(CONFIG_FILE)
    if not raw or raw == "" then return end
    local ok, cfg = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok or not cfg then return end
    if cfg.caseMode then _caseMode = cfg.caseMode end
    if cfg.submitAfter ~= nil then _submitAfter = cfg.submitAfter end
    if cfg.submitDelay ~= nil then _submitDelay = cfg.submitDelay end
    if cfg.autoSubmit ~= nil then _autoSubmit = cfg.autoSubmit end
    if cfg.retypeInvalid ~= nil then _retypeInvalid = cfg.retypeInvalid end
    if cfg.enabled ~= nil then _enabled = cfg.enabled end
    if cfg.blacklist then blacklistText = cfg.blacklist end
    if cfg.keyboardMode ~= nil then _keyboardMode = cfg.keyboardMode and true or false end
    if cfg.uiScale ~= nil then
        local v = tonumber(cfg.uiScale)
        if v then _uiScale = math.clamp(v, 0.6, 3) end
    end
    if cfg.antiRagdoll ~= nil then _antiRagdoll = cfg.antiRagdoll and true or false end
    if cfg.antiLag ~= nil then _antiLag = cfg.antiLag and true or false end
    if cfg.anchor ~= nil then _anchored = cfg.anchor and true or false end
    if cfg.anchored ~= nil then _anchored = cfg.anchored and true or false end
    if cfg.bgIndex ~= nil then
        local v = tonumber(cfg.bgIndex)
        if v and v >= 1 and v <= #BG_ASSET_LIST then
            _bgIndex = math.floor(v)
            BG_IMAGE = "rbxassetid://" .. BG_ASSET_LIST[_bgIndex]
        end
    elseif cfg.bgAssetId then
        for idx, id in ipairs(BG_ASSET_LIST) do
            if tostring(id) == tostring(cfg.bgAssetId) then
                _bgIndex = idx
                BG_IMAGE = "rbxassetid://" .. BG_ASSET_LIST[_bgIndex]
                break
            end
        end
    end
    if cfg.toggleKey then
        local ok, key = pcall(function() return Enum.KeyCode[cfg.toggleKey] end)
        if ok and key then _toggleKey = key end
    end
    if cfg.submitKey then
        local ok, key = pcall(function() return Enum.KeyCode[cfg.submitKey] end)
        if ok and key then _submitKey = key end
    end
    if cfg.autoBuyKey then
        local ok, key = pcall(function() return Enum.KeyCode[cfg.autoBuyKey] end)
        if ok and key then _autoBuyKey = key end
    end
    if cfg.autoBuy ~= nil then _autoBuy = cfg.autoBuy and true or false end
    return cfg
end

local function _arResetCharacter(char)
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root or hum.Health <= 0 then return end
    pcall(function()
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        hum:ChangeState(Enum.HumanoidStateType.Running)
        root.Velocity = Vector3.zero
        root.RotVelocity = Vector3.zero
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        hum.PlatformStand = false
        hum.Sit = false
        hum.AutoRotate = true
        hum.JumpPower = hum.JumpPower > 0 and hum.JumpPower or 50
        hum.WalkSpeed = hum.WalkSpeed > 0 and hum.WalkSpeed or 16
        for _, obj in ipairs(char:GetDescendants()) do
            if obj:IsA("Motor6D") then
                obj.Enabled = true
            elseif obj:IsA("Constraint") or obj:IsA("BallSocketConstraint") or obj:IsA("HingeConstraint") then
                obj.Enabled = true
            elseif obj:IsA("BasePart") then
                obj.CanCollide = true
                obj.AssemblyLinearVelocity = Vector3.zero
                obj.AssemblyAngularVelocity = Vector3.zero
            end
        end
        workspace.CurrentCamera.CameraSubject = hum
        local PM = lp:FindFirstChild("PlayerScripts") and lp.PlayerScripts:FindFirstChild("PlayerModule")
        if PM then
            local CM = PM:FindFirstChild("ControlModule")
            if CM then
                local success, module = pcall(require, CM)
                if success and module and module.Enable then
                    module:Enable()
                end
            end
        end
    end)
end
local function startAntiRagdoll()
    if antiRagdollConn then return end
    antiRagdollConn = RunService.Heartbeat:Connect(function()
        if not _antiRagdoll then return end
        local char = lp.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        local state = hum:GetState()
        if state == Enum.HumanoidStateType.Physics or
           state == Enum.HumanoidStateType.Ragdoll or
           state == Enum.HumanoidStateType.FallingDown or
           state == Enum.HumanoidStateType.Dead or
           hum.PlatformStand == true or
           hum.Sit == true then
            _arResetCharacter(char)
        end
    end)
    if not antiRagdollRespawnConn then
        antiRagdollRespawnConn = lp.CharacterAdded:Connect(function(char)
            task.defer(function()
                if _antiRagdoll and not antiRagdollConn then
                    startAntiRagdoll()
                end
            end)
        end)
    end
end
local function stopAntiRagdoll()
    if antiRagdollConn then
        pcall(function() antiRagdollConn:Disconnect() end)
        antiRagdollConn = nil
    end
    if antiRagdollRespawnConn then
        pcall(function() antiRagdollRespawnConn:Disconnect() end)
        antiRagdollRespawnConn = nil
    end
end
local function applyAntiLagDerender(obj)
    pcall(function()
        if obj:IsA("Accessory") or obj:IsA("Hat") then
            obj:Destroy()
        elseif obj:IsA("BasePart") then
            obj.Material = Enum.Material.Plastic
            obj.Reflectance = 0
            obj.CastShadow = false
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 1
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
            obj.Enabled = false
        elseif obj:IsA("AnimationController") or obj:IsA("Animator") then
            for _, t in ipairs(obj:GetPlayingAnimationTracks()) do
                pcall(function() t:Stop(0) end)
            end
        end
    end)
end
local function enableAntiLag()
    removeAccessoriesEnabled = true
    _antiLag = true
    defLightBrightness = defLightBrightness or Lighting.Brightness
    defLightClock = defLightClock or Lighting.ClockTime
    defLightAmbient = defLightAmbient or Lighting.OutdoorAmbient
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 1e10
    Lighting.Brightness = 1
    Lighting.EnvironmentDiffuseScale = 0
    Lighting.EnvironmentSpecularScale = 0
    for _, e in pairs(Lighting:GetChildren()) do
        pcall(function()
            if e:IsA("BlurEffect") or e:IsA("SunRaysEffect") or e:IsA("ColorCorrectionEffect") or
               e:IsA("BloomEffect") or e:IsA("DepthOfFieldEffect") then
                e.Enabled = false
            end
        end)
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        applyAntiLagDerender(obj)
    end
    if antiLagDescConn then antiLagDescConn:Disconnect() end
    antiLagDescConn = workspace.DescendantAdded:Connect(function(obj)
        if removeAccessoriesEnabled then applyAntiLagDerender(obj) end
    end)
end
local function disableAntiLag()
    removeAccessoriesEnabled = false
    _antiLag = false
    if antiLagDescConn then
        antiLagDescConn:Disconnect()
        antiLagDescConn = nil
    end
    pcall(function()
        if defLightBrightness then Lighting.Brightness = defLightBrightness end
        if defLightClock then Lighting.ClockTime = defLightClock end
        if defLightAmbient then Lighting.OutdoorAmbient = defLightAmbient end
    end)
end

local function startRebind(which)
    _rebinding = which
    if which == "toggle" and ToggleKeyBtn then
        ToggleKeyBtn.Text = "PRESS KEY"
    elseif which == "submit" and SubmitKeyBtn then
        SubmitKeyBtn.Text = "PRESS KEY"
    elseif which == "autobuy" and AutoBuyKeyBtn then
        AutoBuyKeyBtn.Text = "PRESS KEY"
    end
    if setStatus then
        local label = (which == "toggle" and "Code sniper toggle")
            or (which == "submit" and "Manual submit")
            or "Auto buy"
        setStatus("Press any key for " .. label, T.LightCyan, true)
    end
end

local function finishRebind(key)
    if not _rebinding then return end
    local which = _rebinding
    if not key or key == Enum.KeyCode.Unknown then
        _rebinding = nil
        refreshKeyButtons()
        return
    end
    if key == Enum.KeyCode.Escape then
        _rebinding = nil
        refreshKeyButtons()
        if setStatus then setStatus("Keybind cancelled", T.Yellow, true) end
        return
    end
    if which == "toggle" then
        _toggleKey = key
    elseif which == "submit" then
        _submitKey = key
    elseif which == "autobuy" then
        _autoBuyKey = key
    end
    _rebinding = nil
    refreshKeyButtons()
    saveConfig()
    if setStatus then
        local label = (which == "toggle" and "Code sniper toggle")
            or (which == "submit" and "Manual submit")
            or "Auto buy"
        setStatus(label .. " key -> " .. keyName(key), T.LightCyan, true)
    end
end

local WIN_HOME      = UDim2.new(1, -10, 0, 10)

local function clampToScreen(frame, x, y)
    local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1280, 720)
    local size = frame.AbsoluteSize
    local anchor = frame.AnchorPoint
    local left = x - size.X * anchor.X
    local top  = y - size.Y * anchor.Y
    local minX, maxX = 0, math.max(0, vp.X - size.X)
    local minY, maxY = 0, math.max(0, vp.Y - size.Y)
    left = math.clamp(left, minX, maxX)
    top  = math.clamp(top,  minY, maxY)
    return left + size.X * anchor.X, top + size.Y * anchor.Y
end
local function resetPositions()
    Win.Position = WIN_HOME
    SettingsFrame.Position = SETTINGS_HOME
end
applyUIScale = function(v)
    v = math.clamp(tonumber(v) or _uiScale or 0.86, 0.6, 3)
    _uiScale = math.floor(v * 100 + 0.5) / 100
    if MainScale then MainScale.Scale = _uiScale end
    if SettingsScale then SettingsScale.Scale = _uiScale end
    if ScaleInput then ScaleInput.Text = string.format("%.2f", _uiScale) end
    saveConfig()
    task.defer(function()
        if Win and Win.Parent then
            local wx, wy = clampToScreen(Win, Win.AbsolutePosition.X + Win.AbsoluteSize.X * Win.AnchorPoint.X,
                                              Win.AbsolutePosition.Y + Win.AbsoluteSize.Y * Win.AnchorPoint.Y)
            Win.Position = UDim2.new(0, wx, 0, wy)
        end
        if SettingsFrame and SettingsFrame.Parent then
            local sx, sy = clampToScreen(SettingsFrame,
                SettingsFrame.AbsolutePosition.X + SettingsFrame.AbsoluteSize.X * SettingsFrame.AnchorPoint.X,
                SettingsFrame.AbsolutePosition.Y + SettingsFrame.AbsoluteSize.Y * SettingsFrame.AnchorPoint.Y)
            SettingsFrame.Position = UDim2.new(0, sx, 0, sy)
        end
    end)
end

            drag = true

    SettingsFrame.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then

    end)
    UserInputService.InputChanged:Connect(function(inp)
        if drag and (inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch) then
            local d = inp.Position - ds
            local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1280, 720)
            local wantX = dp.X.Scale * vp.X + dp.X.Offset + d.X
            local wantY = dp.Y.Scale * vp.Y + dp.Y.Offset + d.Y
            local cx, cy = clampToScreen(SettingsFrame, wantX, wantY)
            SettingsFrame.Position = UDim2.new(0, cx, 0, cy)

updateSniperToggleUI = function(animate)
    if not OnBtn then return end
    local targetBg = _enabled and T.Green or Color3.fromRGB(65, 65, 80)
    local targetKnobBg = _enabled and Color3.fromRGB(0, 0, 0) or T.White
    local targetPos = _enabled and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
    if animate and TweenService then
        pcall(function()
            TweenService:Create(OnBtn, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = targetBg}):Play()
            if OnBtnKnob then
                TweenService:Create(OnBtnKnob, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = targetKnobBg, Position = targetPos}):Play()
            end
        end)
    else
        OnBtn.BackgroundColor3 = targetBg
        if OnBtnKnob then
            OnBtnKnob.BackgroundColor3 = targetKnobBg
            OnBtnKnob.Position = targetPos
        end
    end
end

UserInputService.InputBegan:Connect(function(input, gp)
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if _rebinding then
        finishRebind(input.KeyCode)
        return
    end
    if gp then return end

    local focused = UserInputService:GetFocusedTextBox()
    if focused and GUI and focused:IsDescendantOf(GUI) then return end

    if input.KeyCode == _toggleKey then
        _enabled = not _enabled
        saveConfig()
        updateSniperToggleUI(true)
        if _enabled then
            setStatus(READY_TEXT(), T.Green, true)
        else
            setStatus("Paused", T.Dim, true)
        end
        return
    end

    if input.KeyCode == _submitKey then
        if pressEnter then
            local ok = pressEnter("keybind")
            if ok then
                setStatus("Manual submit fired", T.LightCyan, true)
            end
        end
        return
    end

    if input.KeyCode == _autoBuyKey then
        toggleAutoBuy()
        return
    end
end)

    local bText = BInput.Text
    blacklistText = bText
    blacklist = {}
    for word in bText:gmatch("[^,]+") do
        local w = word:gsub("%s+", "")
        if w ~= "" then table.insert(blacklist, w:lower()) end
    end
    saveConfig()
end

rememberPendingSubmission = function(box, text, replaceExisting)
    if not _retypeInvalid or not text or text == "" then return end
    if not replaceExisting
    and _pendingRejectedText
    and tick() <= _pendingRejectedUntil then
        return
    end
    _pendingRejectedToken = _pendingRejectedToken + 1
    local token = _pendingRejectedToken
    _pendingRejectedText = text
    _pendingRejectedBox = box
    _pendingRejectedUntil = tick() + 8
    task.delay(8, function()
        if token == _pendingRejectedToken then
            clearPendingSubmission()
        end
    end)
end
restoreRejectedText = function(box, previousText)
    if not _retypeInvalid or not previousText or previousText == "" then
        return false
    end
    local repasteBox = _focused
    if (not repasteBox or not repasteBox.Parent) and box and box.Parent then
        repasteBox = box
    end
    if not repasteBox or not repasteBox.Parent then
        return false
    end
    local restored = pcall(function()
        repasteBox.ClearTextOnFocus = false
        repasteBox.TextEditable = true
        repasteBox.Text = previousText
        repasteBox.CursorPosition = #previousText + 1
    end)
    if restored then
        _focused = repasteBox
        _lastNonBlankBoxText = previousText
        flashCode(previousText, T.Yellow)
    end
    return restored
end
handleRedemptionFeedback = function(text, feedbackObject)
    if not _retypeInvalid or not _pendingRejectedText then return end
    if tick() > _pendingRejectedUntil then
        clearPendingSubmission()
        return
    end
    if feedbackObject and GUI and feedbackObject:IsDescendantOf(GUI) then return end
    local lower = tostring(text or ""):lower()
    local rejected = lower:find("invalid code", 1, true)
        or lower:find("code is invalid", 1, true)
        or lower:find("expired", 1, true)
        or lower:find("already redeemed", 1, true)
        or lower:find("already used", 1, true)
        or lower:find("doesn't exist", 1, true)
        or lower:find("does not exist", 1, true)
        or lower:find("not found", 1, true)
        or lower:find("rejected", 1, true)
        or lower:find("incorrect code", 1, true)
        or lower:find("wrong code", 1, true)
    if not rejected then return end
    cancelRedeemSuccess()
    local previousText = _pendingRejectedText
    local previousBox = _pendingRejectedBox

local function processCode(code, tDetect)
    if not _enabled then return end
    if not code or type(code) ~= "string" or code == "" then return end
    if isFillerWord(code) then return end
    if not isCaseAllowed(code) then return end
    if isWordBlacklisted(code) then return end
    if isUserFiltered(code) then return end
    if codeSeenAt[code] == nil then codeSeenAt[code] = tick() end
    pushRecent(formatCode(code))
    if typeIntoBox(code) then
        local progress = getRealTokenCount()
        local targetWords = _submitAfter or 3
        local detectElapsed = tDetect and (os.clock() - tDetect) or 0
        local _detectMsg = string.format("Detected %d/%d -> %s (%s seconds)",
            progress, targetWords, code, fmtSpeedSeconds(detectElapsed))
        local ready = (progress >= targetWords)
        if ready then
            if _pendingTimer then pcall(task.cancel, _pendingTimer) end
            if _autoSubmit then
                local capturedPartsSnapshot = table.concat(_capturedParts)
                local codeSnapshot = code
                local function submitNow()
                    _pendingTimer = nil
                    local box = aceCodeBox() or _focused
                    local submitText
                    if _keyboardMode then
                        submitText = (box and box.Text) or _lastNonBlankBoxText or formatCode(codeSnapshot)
                    else
                        submitText = capturedPartsSnapshot
                        if submitText == "" then
                            submitText = (box and box.Text) or formatCode(codeSnapshot)
                        end
                        if box and isBoxStillOpen(box) and (box.Text or "") ~= submitText then
                            pcall(function() box.Text = submitText end)
                        end
                    end
                    local now = tick()
                    if submitText ~= "" and _lastAutoSubmitText == submitText and (now - _lastAutoSubmitAt) < 1.5 then
                        codeSeenAt[codeSnapshot] = nil
                        resetPasteCounter()
                        return
                    end
                    _lastAutoSubmitText = submitText
                    _lastAutoSubmitAt = now
                    local fired = pressEnter("auto", submitText)
                    if fired then
                        setStatus("Successfully Auto submitted!", T.LightCyan, true)
                    else
                        local retryBox = aceCodeBox() or box or _focused
                        if retryBox and isBoxStillOpen(retryBox) then
                            syncSubmitText(retryBox, submitText)
                            if pressEnter("auto", submitText) then
                                setStatus("Successfully Auto submitted!", T.LightCyan, true)
                            else
                                
                            end
                        else
                            
                        end
                    end
                    codeSeenAt[codeSnapshot] = nil
                    resetPasteCounter()
                end
                submitNow()
                return
            else
                copyBoxText()
                resetPasteCounter()
                setStatus("Auto submit is off", T.Yellow)
            end
        end

        setStatus(_detectMsg, T.Green)
    end
end

local function isAceAnnouncement(...)
    local args = table.pack(...)
    if args.n == 0 or typeof(args[1]) ~= "string" then return false end
    for index = 2, args.n do
        local value = args[index]
        if typeof(value) == "string"
        and (value:find("Sounds%.")
            or value:find("rbxassetid")
            or ACE_POSITIONS[value]) then
            return true
        end
    end
    return false
end
local function aceStripRich(text)
    if type(text) ~= "string" then return tostring(text) end
    return (text:gsub("<[^>]->", ""))
end
local function isTimerLikeToken(raw)
    if not raw or raw == "" then return true end
    local w = tostring(raw)
    if w:find(":") then return true end
    local clean = w:gsub("[^A-Za-z0-9]", "")
    if clean == "" then return true end
    local lower = clean:lower()
    if clean:match("^%d+[smhdSMHD]$") then return true end
    if clean:match("^%d+[mM]in") then return true end
    if clean:match("^%d+[sS]ec") then return true end
    if clean:match("^%d+[hH]r") then return true end
    if clean:match("^%d+$") and #clean >= 4 then return true end
    if lower:match("^transparency") then return true end
    if lower:match("^color") then return true end
    return false
end

local function isFillerWord(rawWord)
    if not rawWord or rawWord == "" then return true end
    local clean = tostring(rawWord):gsub("[^A-Za-z0-9]", ""):lower()
    if clean == "" then return true end
    if #clean == 1 and clean:match("[a-z]") then return true end
    if FILLER_WORDS[clean] then return true end
    return false
end

local function aceTokenize(text)
    local words = {}
    for word in text:gmatch("%S+") do
        if not isTimerLikeToken(word) then
            local clean = word:gsub("[^A-Za-z0-9_]", "")
            if clean ~= "" and not isWordBlacklisted(clean) and not isFillerWord(clean) then
                words[#words + 1] = clean
            end
        end
    end
    return words
end
local function onAceAnnouncement(...)

    local tDetect = os.clock()
    local text = aceStripRich(tostring((...) or ""))
    if text == "" then return end
    local tokens = aceTokenize(text)
    if #tokens == 0 then return end
    for _, word in ipairs(tokens) do
        local captured = formatCode(word)
        if captured ~= "" and not _seen[captured] then
            _seen[captured] = true

            task.defer(function() _seen[captured] = nil end)
            processCode(captured, tDetect)
        end
    end
end

RunService.Heartbeat:Connect(function(dt)
    if not _autoBuy then return end
    autoBuyAccumulator = autoBuyAccumulator + dt
    if autoBuyAccumulator < AUTO_BUY_INTERVAL then return end
    autoBuyAccumulator = 0

    local total = #autoBuyPrompts
    if total == 0 then return end

    local processed = 0
    while processed < AUTO_BUY_BATCH and total > 0 do
        if autoBuyIndex > #autoBuyPrompts then autoBuyIndex = 1 end
        local prompt = autoBuyPrompts[autoBuyIndex]
        if not prompt or not prompt.Parent or not autoBuyPromptSet[prompt] then
            table.remove(autoBuyPrompts, autoBuyIndex)
            total = #autoBuyPrompts
        else
            pcall(function()
                prompt.HoldDuration = 0
                prompt:InputHoldBegin()
                prompt:InputHoldEnd()
            end)
            autoBuyIndex = autoBuyIndex + 1
            processed = processed + 1
        end
    end
end)
